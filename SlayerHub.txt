local PREFIX = "[XenoPreflight]"
local pass, fail, present, missing = 0, 0, 0, 0
local results = {}

local function log(...)
    print(PREFIX, ...)
end

local function record(name, state, detail)
    results[name] = {state = state, detail = tostring(detail or "")}
    if state == "PASS" then pass += 1
    elseif state == "FAIL" then fail += 1
    elseif state == "PRESENT" then present += 1
    elseif state == "MISSING" then missing += 1 end
    local fn = (state == "FAIL" or state == "MISSING") and warn or print
    fn(PREFIX, string.format("%-9s %-28s %s", state, name, tostring(detail or "")))
end

local function exists(name, value)
    if typeof(value) == "function" then
        record(name, "PRESENT", "função encontrada")
        return true
    end
    record(name, "MISSING", "não encontrada")
    return false
end

local function test(name, available, callback)
    if not available then
        record(name, "MISSING", "API ausente")
        return false
    end
    local ok, result = pcall(callback)
    if ok and result ~= false then
        record(name, "PASS", result == true and "ok" or result)
        return true
    end
    record(name, "FAIL", ok and "teste retornou false" or result)
    return false
end

local hookFn = typeof(hookfunction) == "function" and hookfunction
    or (typeof(replaceclosure) == "function" and replaceclosure or nil)
local identifyFn = typeof(identifyexecutor) == "function" and identifyexecutor
    or (typeof(getexecutorname) == "function" and getexecutorname or nil)
local clipboardFn = typeof(setclipboard) == "function" and setclipboard
    or (typeof(toclipboard) == "function" and toclipboard or nil)
local getUpvaluesFn = typeof(getupvalues) == "function" and getupvalues
    or (debug and typeof(debug.getupvalues) == "function" and debug.getupvalues or nil)

log("iniciando...")

if identifyFn then
    local ok, name, version = pcall(identifyFn)
    if ok then
        log("Executor:", tostring(name), tostring(version or "versão não informada"))
    else
        record("identifyexecutor", "FAIL", name)
    end
else
    record("identifyexecutor", "MISSING", "identifyexecutor/getexecutorname ausente")
end

test("loadstring", typeof(loadstring) == "function", function()
    local fn, err = loadstring("return 40 + 2")
    assert(typeof(fn) == "function", tostring(err))
    assert(fn() == 42, "retorno incorreto")
    return "compilação + execução ok"
end)

test("getgc", typeof(getgc) == "function", function()
    local t = getgc()
    assert(type(t) == "table", "não retornou table")
    assert(#t > 0, "table vazia")
    return tostring(#t) .. " objetos"
end)

test("getrawmetatable", typeof(getrawmetatable) == "function", function()
    local mt = {__metatable = "locked"}
    local obj = setmetatable({}, mt)
    assert(getrawmetatable(obj) == mt, "metatable incorreta")
    return true
end)

test("setreadonly", typeof(setreadonly) == "function", function()
    local t = {value = 1}
    if typeof(table.freeze) ~= "function" then return "presente; table.freeze indisponível" end
    table.freeze(t)
    setreadonly(t, false)
    t.value = 2
    assert(t.value == 2, "não liberou escrita")
    return true
end)

test("newcclosure", typeof(newcclosure) == "function", function()
    local fn = newcclosure(function() return 123 end)
    assert(typeof(fn) == "function" and fn() == 123, "closure inválida")
    if typeof(iscclosure) == "function" then
        assert(iscclosure(fn) == true, "não foi reconhecida como C closure")
    end
    return true
end)

test("hookfunction", typeof(hookFn) == "function", function()
    local function original() return 1 end
    local old = hookFn(original, function() return 2 end)
    assert(typeof(old) == "function", "não retornou função original")
    assert(original() == 2, "hook não alterou a função")
    assert(old() == 1, "referência original incorreta")
    return true
end)

test("getconnections", typeof(getconnections) == "function", function()
    local ev = Instance.new("BindableEvent")
    local rbxc = ev.Event:Connect(function() end)
    local list = getconnections(ev.Event)
    local c = type(list) == "table" and list[1] or nil
    assert(c ~= nil, "nenhuma conexão retornada")
    -- SlayerHub usa c.Function; se Xeno não expõe Function, o recurso específico será ignorado.
    local hasFunctionField = c.Function ~= nil
    rbxc:Disconnect()
    ev:Destroy()
    return hasFunctionField and "Function field disponível" or "básico ok; Function field ausente"
end)

test("getupvalues/debug.getupvalues", typeof(getUpvaluesFn) == "function", function()
    local marker = {}
    local function holder() return marker end
    local values = getUpvaluesFn(holder)
    assert(type(values) == "table", "não retornou table")
    return true
end)

test("gethui", typeof(gethui) == "function", function()
    local hui = gethui()
    assert(typeof(hui) == "Instance", "não retornou Instance")
    return hui:GetFullName()
end)

-- Testa se a UI embutida consegue parentear uma ScreenGui; remove imediatamente.
test("UI parent", true, function()
    local parent
    if typeof(gethui) == "function" then
        local ok, hui = pcall(gethui)
        if ok and typeof(hui) == "Instance" then parent = hui end
    end
    if not parent then
        local ok, core = pcall(game.GetService, game, "CoreGui")
        if ok then parent = core end
    end
    if not parent then
        parent = game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui", 10)
    end
    assert(parent, "nenhum parent de UI disponível")
    local gui = Instance.new("ScreenGui")
    gui.Name = "__XenoPreflightUI"
    gui.ResetOnSpawn = false
    gui.Parent = parent
    task.wait()
    assert(gui.Parent == parent, "ScreenGui não permaneceu parentada")
    gui:Destroy()
    return parent:GetFullName()
end)
